function Wait-TxFinal([string]$id) {
    for ($i=0; $i -lt 45; $i++) {
        try {
            $raw = & curl.exe --silent --show-error --max-time 8 "$base/transfers/requests/$id"
            if ($LASTEXITCODE -eq 0 -and $raw) {
                $result = $raw | ConvertFrom-Json
                if ($result.status -in @('COMMITTED','ABORTED')) { return $result }
            }
        } catch { }
        Start-Sleep -Seconds 2
    }
    throw "Transaction outcome unresolved: $id"
}

function Invoke-CheckedLabCompose([string]$Project, [string[]]$Commands) {
    & docker compose -p $Project @Commands
    if ($LASTEXITCODE -ne 0) { throw "Compose failed in $Project : $($Commands -join ' ')" }
}
function Wait-ShardCTcpReady([string]$Project) {
    for ($i = 0; $i -lt 60; $i++) {
        $probe = & docker compose -p $Project --profile expansion exec -T shard-c-db pg_isready -h 127.0.0.1 -p 5432 -U banking -d shard_c 2>&1
        if ($LASTEXITCODE -eq 0) {
            Write-Host "SHARD C TCP READY: $Project"
            return
        }
        Start-Sleep -Seconds 2
    }
    throw "Shard C PostgreSQL did not accept TCP connections in $Project. Preserve this lab for inspection."
}
function Wait-Stage14Fault([string]$Project, [string]$Point) {
    for ($i = 0; $i -lt 25; $i++) {
        $sql = "SELECT COALESCE((SELECT state::text FROM stage14_fault_experiments WHERE point='$Point' ORDER BY created_at DESC LIMIT 1),'MISSING');"
        $state = (& docker compose -p $Project exec -T metadata-db psql -U banking -d metadata -t -A -c $sql) | Out-String
        if ($LASTEXITCODE -ne 0) { throw "Cannot query fault state in $Project" }
        if ($state.Trim() -eq 'FIRED') {
            Write-Host "CONFIRMED FIRED: $Point in $Project"
            return
        }
        Start-Sleep -Seconds 2
    }
    throw "Fault was not FIRED: $Point in $Project. Preserve lab for investigation."
}
function Wait-Stage14CutoverRecovery([string]$Project) {
    for ($i = 0; $i -lt 30; $i++) {
        $sql = "SELECT phase FROM migration_jobs WHERE source_shard='B' AND target_shard='C' ORDER BY created_at DESC LIMIT 1;"
        $phase = (& docker compose -p $Project exec -T metadata-db psql -U banking -d metadata -t -A -c $sql) | Out-String
        if ($LASTEXITCODE -ne 0) { throw "Cannot query recovery phase in $Project" }
        Write-Host "RECOVERY PHASE: $($phase.Trim())"
        if ($phase.Trim() -in @('VERIFYING','COMPLETED')) { return }
        Start-Sleep -Seconds 3
    }
    throw "Cutover recovery has not completed in $Project. Preserve lab for investigation."
}
function Invoke-Stage14Lab {
    param(
        [Parameter(Mandatory=$true)]
        [ValidateSet('SNAPSHOT_PERSISTED','DELTA_GROUP_APPLIED','FENCING_GATE_CLOSED','MAP_ACTIVATED')]
        [string]$Point,
        [Parameter(Mandatory=$true)]
        [int]$Port
    )
    $slug = @{
        SNAPSHOT_PERSISTED = 'snapshot'
        DELTA_GROUP_APPLIED = 'delta'
        FENCING_GATE_CLOSED = 'fencing'
        MAP_ACTIVATED = 'map'
    }[$Point]
    $suffix = [guid]::NewGuid().ToString('N').Substring(0,8)
    $project = "banking-v-$slug-$suffix"
    $labBase = "http://localhost:$Port"
    $saved = @{}
    $keys = @('COMPOSE_PROJECT_NAME','GATEWAY_PORT','STAGE07_ENABLE_FAULTS','STAGE09_ENABLE_FAULTS','STAGE14_ENABLE_FAULTS')
    foreach ($key in $keys) { $saved[$key] = [Environment]::GetEnvironmentVariable($key,'Process') }
    $labSucceeded = $false
    Push-Location 'E:\SharifiProject\sharding\banking-video-ab'
    try {
        if ($project -eq 'banking-video-ab') { throw 'Refusing to use the main demo project' }
        $portInUse = @(Get-NetTCPConnection -LocalPort $Port -State Listen -ErrorAction SilentlyContinue)
        if ($portInUse.Count -gt 0) { throw "Port $Port is in use. Use a free port for this lab." }
        $existingContainers = @(docker ps -a -q --filter "label=com.docker.compose.project=$project" | Where-Object { $_ })
        $existingVolumes = @(docker volume ls -q --filter "label=com.docker.compose.project=$project" | Where-Object { $_ })
        $existingNetworks = @(docker network ls -q --filter "label=com.docker.compose.project=$project" | Where-Object { $_ })
        if ($existingContainers.Count -gt 0 -or $existingVolumes.Count -gt 0 -or $existingNetworks.Count -gt 0) {
            throw "Not a fresh project: $project"
        }
        $env:COMPOSE_PROJECT_NAME = $project
        $env:GATEWAY_PORT = [string]$Port
        $env:STAGE07_ENABLE_FAULTS = '0'
        $env:STAGE09_ENABLE_FAULTS = '0'
        $env:STAGE14_ENABLE_FAULTS = '1'
        Write-Host "ACTIVE ISOLATED LAB: $project"
        Write-Host "GATEWAY: $labBase"
        Invoke-CheckedLabCompose $project @('config','--quiet')
        Invoke-CheckedLabCompose $project @('up','-d','--build','metadata-db','shard-a-db','shard-b-db','schema-migrator','shard-a-1','shard-a-2','shard-b-1','shard-b-2','coordinator','gateway','self-healing')
        Invoke-CheckedLabCompose $project @('--profile','expansion','up','-d','shard-c-db')
        Wait-ShardCTcpReady $project
        Invoke-CheckedLabCompose $project @('--profile','expansion','run','--rm','--no-deps','schema-migrator','python','-m','scripts.demo_migrate','--targets','shard-c')
        Invoke-CheckedLabCompose $project @('--profile','expansion','up','-d','--no-deps','shard-c-1','shard-c-2','migration-manager')
        Invoke-CheckedLabCompose $project @('--profile','expansion','run','--rm','--no-deps','schema-migrator','python','-m','scripts.demo_enable_c')
        Invoke-CheckedLabCompose $project @('exec','-T','metadata-db','psql','-U','banking','-d','metadata','-c',"SELECT job_id,phase FROM migration_jobs ORDER BY created_at DESC LIMIT 3;")
        if ($Point -eq 'SNAPSHOT_PERSISTED') {
            Invoke-CheckedLabCompose $project @('run','--rm','--no-deps','schema-migrator','python','-m','scripts.stage14_fault','--action','plan')
            Invoke-CheckedLabCompose $project @('run','--rm','--no-deps','-e','STAGE14_ENABLE_FAULTS=1','schema-migrator','python','-m','scripts.stage14_fault','--action','arm','--point',$Point)
            Invoke-CheckedLabCompose $project @('run','--rm','--no-deps','schema-migrator','python','-m','scripts.stage14_fault','--action','trigger','--point',$Point)
        } elseif ($Point -eq 'DELTA_GROUP_APPLIED') {
            Invoke-CheckedLabCompose $project @('run','--rm','--no-deps','schema-migrator','python','-m','scripts.verify_stage10','--scenario=bootstrap')
            Invoke-CheckedLabCompose $project @('run','--rm','--no-deps','schema-migrator','python','-m','scripts.verify_stage11','--scenario=bootstrap')
            Invoke-CheckedLabCompose $project @('--profile','expansion','stop','migration-manager')
            $depositId = 'video-delta-in-' + [guid]::NewGuid().ToString('N')
            $depositResult = Invoke-RestMethod "$labBase/accounts/A2000/deposit" -Method Post -ContentType application/json -Body (@{amount='1.00';request_id=$depositId} | ConvertTo-Json)
            if ($depositResult.status -ne 'COMMITTED') { throw "Deposit not committed: $depositId" }
            $withdrawId = 'video-delta-out-' + [guid]::NewGuid().ToString('N')
            $withdrawResult = Invoke-RestMethod "$labBase/accounts/A2000/withdraw" -Method Post -ContentType application/json -Body (@{amount='1.00';request_id=$withdrawId} | ConvertTo-Json)
            if ($withdrawResult.status -ne 'COMMITTED') { throw "Withdrawal not committed: $withdrawId" }
            Write-Host "SOURCE WRITES COMMITTED: $depositId $withdrawId"
            Invoke-CheckedLabCompose $project @('run','--rm','--no-deps','-e','STAGE14_ENABLE_FAULTS=1','schema-migrator','python','-m','scripts.stage14_fault','--action','arm','--point',$Point)
            Invoke-CheckedLabCompose $project @('--profile','expansion','start','migration-manager')
        } else {
            Invoke-CheckedLabCompose $project @('run','--rm','--no-deps','schema-migrator','python','-m','scripts.verify_stage10','--scenario=bootstrap')
            Invoke-CheckedLabCompose $project @('run','--rm','--no-deps','schema-migrator','python','-m','scripts.verify_stage11','--scenario=bootstrap')
            Invoke-CheckedLabCompose $project @('run','--rm','--no-deps','schema-migrator','python','-m','scripts.verify_stage11','--scenario=online')
            Invoke-CheckedLabCompose $project @('run','--rm','--no-deps','schema-migrator','python','-m','scripts.verify_stage12','--scenario=preflight')
            Invoke-CheckedLabCompose $project @('run','--rm','--no-deps','-e','STAGE14_ENABLE_FAULTS=1','schema-migrator','python','-m','scripts.stage14_fault','--action','arm','--point',$Point)
            Invoke-CheckedLabCompose $project @('run','--rm','--no-deps','schema-migrator','python','-m','scripts.stage14_fault','--action','trigger','--point',$Point)
        }
        Wait-Stage14Fault $project $Point
        Invoke-CheckedLabCompose $project @('--profile','expansion','ps','-a','migration-manager')
        Invoke-CheckedLabCompose $project @('run','--rm','--no-deps','schema-migrator','python','-m','scripts.stage14_fault','--action','inspect')
        Invoke-CheckedLabCompose $project @('run','--rm','--no-deps','schema-migrator','python','-m','scripts.stage14_fault','--action','verify','--point',$Point,'--moment','crashed')
        $env:STAGE14_ENABLE_FAULTS = '0'
        Invoke-CheckedLabCompose $project @('--profile','expansion','up','-d','--no-deps','--force-recreate','migration-manager')
        if ($Point -eq 'SNAPSHOT_PERSISTED') {
            Invoke-CheckedLabCompose $project @('run','--rm','--no-deps','schema-migrator','python','-m','scripts.verify_stage10','--scenario=bootstrap')
        } elseif ($Point -eq 'DELTA_GROUP_APPLIED') {
            Invoke-CheckedLabCompose $project @('run','--rm','--no-deps','schema-migrator','python','-m','scripts.verify_stage11','--scenario=resume')
        } else {
            Wait-Stage14CutoverRecovery $project
        }
        Invoke-CheckedLabCompose $project @('run','--rm','--no-deps','schema-migrator','python','-m','scripts.stage14_fault','--action','verify','--point',$Point,'--moment','recovered')
        $labSucceeded = $true
        Write-Host "SUCCESS: $Point crash and recovery verified in $project"
        Invoke-CheckedLabCompose $project @('--profile','expansion','stop')
        Write-Host "LAB STOPPED WITHOUT DELETING DATA: $project"
    } finally {
        foreach ($key in $keys) { [Environment]::SetEnvironmentVariable($key,$saved[$key],'Process') }
        Pop-Location
        if (-not $labSucceeded) {
            Write-Host "LAB NOT COMPLETED; NO MAIN PROJECT COMMANDS WERE RUN. Inspect only: $project" 
            Write-Host "docker compose -p $project --profile expansion ps -a"
        }
    }
}
